---getTotalEnergyForPowerDeviceDaily----
SELECT d.deviceTypeName              as name,
       e.dayWise                     as timePeriod,
       ISNULL(sum(e.totalEnergy), 0) AS totalEnergy
FROM (SELECT DISTINCT dt2.name as deviceTypeName,
                      d.name      deviceName
      FROM device d
               join device_type dt2 on d.device_type_id = dt2.id
      WHERE d.unit_id = :unitId
        AND d.IS_ACTIVE = 1
        AND d.input_type = :inputType
        AND d.device_type_id IN (SELECT id
                                 FROM device_type
                                 WHERE id in (:deviceTypeIds))) d

         LEFT JOIN (SELECT d.name,
                           FORMAT(dr.target_date, 'dd-MMM-yyyy') AS dayWise,
                           CASE
                               WHEN dr.max_energy - dr.min_energy > 10000000 THEN 260
                               ELSE dr.max_energy - dr.min_energy
                               END                           AS totalEnergy
                    FROM daily_consumption_data dr
                             JOIN device d ON d.id = dr.device_id
                             JOIN device_type dt ON d.device_type_id = dt.id
                    WHERE dr.target_date between :fromDate + ' 12:00:00.0000000 AM' and :toDate + ' 11:59:59.0000000 PM'
                      AND dt.parent_id = 101
                      AND d.unit_id = :unitId) e ON e.name = d.deviceName
group by d.deviceTypeName, e.dayWise
ORDER BY e.dayWise

;

                           CASE
                               WHEN dr.max_energy - dr.min_energy > 10000000 THEN 260
                               ELSE dr.max_energy - dr.min_energy
                               END AS totalEnergy
                    FROM monthly_consumption_data dr
                             JOIN device d ON d.id = dr.device_id
                             JOIN device_type dt ON d.device_type_id = dt.id
                    WHERE dr.target_month between :year + '-01-01' and :year + '-12-31'

                      AND d.unit_id = :unitId
) e ON e.name = d.deviceName
group by d.deviceTypeName, e.monthWise
ORDER BY e.monthWise
;
